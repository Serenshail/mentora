    int* arr = (int*)calloc(numsSize, sizeof(int));

    if(k == 0)

四、memcpy()：为什么还要复制一次？

基本语法：

五、这段代码存在什么问题？
这部分我希望你认真看一下。

代码虽然实现了数组旋转，但还不够严谨。

有四个值得改进的地方。

问题 1：内存泄漏
原代码：

int* arr = (int*)calloc(numsSize, sizeof(int));

k %= numsSize;

if(k == 0)
    return;

问题就在于：

程序申请了内存，但是直接 return 了。

例如：

numsSize = 5;
k = 10;

申请内存后，发现：

k = 10 % 5 = 0;

于是直接退出。

但是申请的内存没有释放。

正确做法是：

free(arr);

free() 的作用就是释放之前动态申请的内存。

而且正常执行完毕之后，同样需要释放：

memcpy(nums, arr, numsSize * sizeof(int));

free(arr);

问题 2：数组长度为 0
原代码：

k %= numsSize;

如果：

numsSize = 0;

就会发生：

k % 0

这是未定义行为。

所以需要先判断：

if(numsSize <= 0)
    return;

问题 3：负数 k
假设：

k = -2;
numsSize = 5;

那么：

k %= numsSize;

在 C 语言中，结果是：

k = -2;

这样：

(i+k)%numsSize

可能产生负数下标，导致访问数组时出现错误。

如果我们希望负数代表向左旋转，就可以这样处理：

k %= numsSize;

if(k < 0)
    k += numsSize;

问题 4：没有充分处理内存申请失败
原代码：

if(arr != NULL)
{
    // 执行旋转
}

如果内存申请失败，整个程序什么也不做。

对于这个函数来说，可以在申请失败后直接返回，并且明确处理这个情况。

六、我帮你整理一个改进版本
下面这份代码保留原来的算法思路，但是修复了上面的问题。

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void rotate(int* nums, int numsSize, int k)
{
    // 1. 检查数组长度
    if(nums == NULL || numsSize <= 1)
        return;

    // 2. 计算实际旋转次数
    k %= numsSize;

    // 3. 处理负数
    if(k < 0)
        k += numsSize;

    // 4. 不需要旋转，直接返回
    if(k == 0)
        return;

    // 5. 申请辅助数组
    int* arr = calloc(numsSize, sizeof(int));

    // 6. 判断是否申请成功
    if(arr == NULL)
        return;

    // 7. 将原数组元素放到新位置
    for(int i = 0; i < numsSize; i++)
    {
        arr[(i + k) % numsSize] = nums[i];
    }

    // 8. 复制回原数组
    memcpy(nums, arr, numsSize * sizeof(int));

    // 9. 释放内存
    free(arr);
}

这段代码的整体执行逻辑：

程序执行流程
接收数组 nums、长度 numsSize、旋转次数 k

检查数组是否有效

无效则直接返回
计算实际旋转次数

k %= numsSize
申请辅助数组 arr

通过 for 循环移动元素

arr[(i+k)%numsSize] = nums[i]
memcpy() 复制回原数组

free(arr)，释放内存

程序结束

我先告诉你结论：

假设原数组是：

下面一步一步理解为什么这样做是成立的。

一、先理解 reverse() 函数
你图片中的第一段代码：

void reverse(int* nums, int start, int end)
{
    while(start < end)
    {
        int tmp = nums[start];
        nums[start] = nums[end];
        nums[end] = tmp;

        start++;
        end--;
    }
}

这个函数的作用非常简单：

将数组中指定范围内的元素反转。

例如：

nums = [1, 2, 3, 4, 5]

如果调用：

reverse(nums, 0, 4);

就表示反转下标从 0 到 4 的全部元素。

结果：

[5, 4, 3, 2, 1]

四、完整代码应该是什么样子？
把你图片中的 reverse() 和 rotate() 组合起来：

void reverse(int* nums, int start, int end)
{
    while(start < end)
    {
        int tmp = nums[start];
        nums[start] = nums[end];
        nums[end] = tmp;

        start++;
        end--;
    }
}

void rotate(int* nums, int numsSize, int k)
{
    if(nums == NULL || numsSize <= 1)
        return;

    k %= numsSize;

    // 第一次：反转整个数组
    reverse(nums, 0, numsSize - 1);

    // 第二次：反转前 k 个元素
    reverse(nums, 0, k - 1);

    // 第三次：反转剩余元素
    reverse(nums, k, numsSize - 1);
}

注意：这里假设题目要求向右旋转，且 k 为非负数。

如果 k = 0，后两次反转的范围为空，不会进入 while 循环，所以也能正常工作。

五、为什么顺序表需要 capacity？——动态扩容

1. 使用 realloc 扩大空间
C 语言提供了：

realloc()

它的作用是调整之前动态申请的内存空间大小。

基本语法：

realloc(原来的内存地址, 新的内存大小);

例如：

a = realloc(a, 10 * sizeof(int));

意思是：

将原来的数组空间调整为可以存放 10 个整数的大小。

但是这里有一个非常重要的细节：

realloc() 可能在原来的位置直接扩大内存，也可能申请一块新的内存，把旧数据复制过去，再释放旧空间。

所以不能假设原来的地址一定不变。

而且，不能直接这样写：

a = realloc(a, 10 * sizeof(int));

因为如果扩容失败，realloc() 会返回 NULL，直接赋值就可能丢失原来的地址，造成内存泄漏。

正确写法：

int* tmp = realloc(a, 10 * sizeof(int));

if(tmp != NULL)
{
    a = tmp;
    capacity = 10;
}

这样即使扩容失败，原来的 a 也还在。

    if(pos < 0 || pos >= ps->size)
        return;

    for(int i = pos; i < ps->size - 1; i++)
    {
        ps->a[i] = ps->a[i + 1];
    }

    ps->size--;
}

七、把这些操作放在一起：顺序表的完整生命周期
一个完整的顺序表，不只是插入和删除，还需要初始化和销毁。

下面是一个相对完整的 C 语言实现。

#include <stdio.h>
#include <stdlib.h>

typedef int SLDataType;

typedef struct SeqList
{
    SLDataType* a;
    int size;
    int capacity;
} SL;

// 1. 初始化顺序表
void SeqListInit(SL* ps)
{
    ps->a = NULL;
    ps->size = 0;
    ps->capacity = 0;
}

// 2. 检查容量，不够就扩容
void SeqListCheckCapacity(SL* ps)
{
    if(ps->size == ps->capacity)
    {
        int newCapacity = ps->capacity == 0
                          ? 4
                          : ps->capacity * 2;

        SLDataType* tmp = realloc(
            ps->a,
            newCapacity * sizeof(SLDataType)
        );

        if(tmp == NULL)
        {
            printf("内存申请失败！\n");
            exit(EXIT_FAILURE);
        }

        ps->a = tmp;
        ps->capacity = newCapacity;
    }
}

// 3. 尾插
void SeqListPushBack(SL* ps, SLDataType x)
{
    SeqListCheckCapacity(ps);

    ps->a[ps->size] = x;
    ps->size++;
}

// 4. 指定位置插入
void SeqListInsert(SL* ps, int pos, SLDataType x)
{
    if(pos < 0 || pos > ps->size)
        return;

    SeqListCheckCapacity(ps);

    for(int i = ps->size; i > pos; i--)
    {
        ps->a[i] = ps->a[i - 1];
    }

    ps->a[pos] = x;
    ps->size++;
}

// 5. 指定位置删除
void SeqListErase(SL* ps, int pos)
{
    if(pos < 0 || pos >= ps->size)
        return;

    for(int i = pos; i < ps->size - 1; i++)
    {
        ps->a[i] = ps->a[i + 1];
    }

    ps->size--;
}

// 6. 打印顺序表
void SeqListPrint(SL* ps)
{
    for(int i = 0; i < ps->size; i++)
    {
        printf("%d ", ps->a[i]);
    }

    printf("\n");
}

// 7. 销毁顺序表
void SeqListDestroy(SL* ps)
{
    free(ps->a);

    ps->a = NULL;
    ps->size = 0;
    ps->capacity = 0;
}

int main()
{
    SL sl;

    // 初始化
    SeqListInit(&sl);

    // 尾插
    SeqListPushBack(&sl, 10);
    SeqListPushBack(&sl, 20);
    SeqListPushBack(&sl, 30);

    SeqListPrint(&sl);

    // 在下标 1 的位置插入 99
    SeqListInsert(&sl, 1, 99);

    SeqListPrint(&sl);

    // 删除下标 2 的元素
    SeqListErase(&sl, 2);

    SeqListPrint(&sl);

    // 销毁
    SeqListDestroy(&sl);

    return 0;
}

三、重点：SLCheckCapacity() 扩容函数到底做了什么？

第五步：为什么要用 tmp 接收 realloc 的结果？
看这段：

SLDataType* tmp = (SLDataType*)realloc(...);

你可能会问：

为什么不直接写：

ps->a = realloc(ps->a, newCapacity * sizeof(SLDataType));

为什么还要多定义一个 tmp？

这是 C 语言动态内存管理中非常重要的一个细节。

因为 realloc() 可能失败。

如果扩容失败，它会返回：

假设你直接写：

如果失败：

那么原来的数组地址就丢失了。

你原本的数据还在原来的内存中，但你已经找不到它了。

这会造成内存泄漏。

所以老师采用了更安全的写法：

SLDataType* tmp = realloc(ps->a, 32);

这样做：

如果成功，tmp 接收调整后的内存地址。

如果失败，tmp 接收 NULL。

原来的 ps->a 不会因为赋值而丢失。

意思是：

第二行：

等价于：

意思是：

更新顺序表记录的总容量。

如果不写第二行，会发生什么？

实际内存已经可以存储 8 个元素了，但结构体中的 capacity 仍然记录 4。

这样，程序就无法正确知道自己拥有多少空间。

所以扩容成功后，必须同时更新：

数组的首地址（如果发生了搬迁）。

数组的容量。

最后：

printf("扩容成功\n");

只是在控制台输出一条提示，不影响扩容本身。

五、为什么还需要一个 SLCheckCapacity() 函数？
现在看老师的另一个函数：

{
    SLCheckCapacity(ps);

    ps->a[ps->size] = x;
    ps->size++;
}

它叫尾插函数。

意思是在顺序表最后面添加一个元素。

你看，这里并没有写一大堆 realloc()。

而是直接调用：

SLCheckCapacity(ps);

为什么？

因为老师把扩容逻辑独立封装成了一个函数。

这样以后无论是尾插还是头插，还是指定位置插入，只要需要增加元素，都可以调用同一个扩容函数。

六、现在来看头插 SLPushFront()
你第二张图里面，老师写了：

void SLPushFront(SL* ps, SLDataType x)
{

}

你可能会想：

既然尾插只需要：

ps->a[ps->size] = x;
ps->size++;

那头插是不是也可以：

ps->a[0] = x;
ps->size++;

不可以。

为什么？

假设现在：

[10, 20, 30, 40]

你想在最前面插入：

5

如果直接执行：

ps->a[0] = 5;

结果就变成：

[5, 20, 30, 40]

原来的 10 丢失了！

所以我们必须先给新元素腾出位置。

具体怎么做？

把原来的所有元素向后移动一位。

void SLPushFront(SL* ps, SLDataType x)
{
    SLCheckCapacity(ps);

    for(int i = ps->size; i > 0; i--)
    {
        ps->a[i] = ps->a[i - 1];
    }

    ps->a[0] = x;
    ps->size++;
}

逐步看：

初始：

[10, 20, 30, 40]

第一步：

ps->a[4] = ps->a[3];

结果：

[10, 20, 30, 40, 40]

第二步：

ps->a[3] = ps->a[2];

结果：

[10, 20, 30, 30, 40]

第三步：

ps->a[2] = ps->a[1];

结果：

[10, 20, 20, 30, 40]

第四步：

ps->a[1] = ps->a[0];

结果：

[10, 10, 20, 30, 40]

最后：

ps->a[0] = 5;
ps->size++;

结果：

[5, 10, 20, 30, 40]

这里最重要的是：

for(int i = ps->size; i > 0; i--)

为什么 i 要从 size 开始，并且不断减小？

因为我们必须从后往前移动，避免覆盖尚未移动的数据。

这和你之前学过的数组元素移动、旋转数组问题，其实是同一个思想：在连续内存中调整元素的位置，需要考虑移动方向和数据覆盖。

七、继续理解尾删和头删
你第三张图下面还有两个函数：

void SLPopBack(SL* ps)  // 尾删
{

}

void SLPopFront(SL* ps) // 头删
{

}

这两个操作和前面的尾插、头插正好相反。

1. 尾删：SLPopBack()
假设：

[10, 20, 30, 40]

现在删除最后一个元素。

你需要做什么？

直接：

ps->size--;

完整代码：

void SLPopBack(SL* ps)
{
    assert(ps != NULL);
    assert(ps->size > 0);

    ps->size--;
}

为什么这么简单？

因为尾删不需要移动其他元素。

原来：

size = 4;

现在：

size = 3;

有效数据变成：

[10, 20, 30]

注意：原来位置 3 可能还残留着 40，但它已经不属于有效元素了。

2. 头删：SLPopFront()
现在删除第一个元素：

[10, 20, 30, 40]

预期结果：

[20, 30, 40]

如果直接：

ps->size--;

就会得到：

[10, 20, 30]

这显然不是我们想要的。

所以必须把后面的元素全部向前移动。

代码：

void SLPopFront(SL* ps)
{
    assert(ps != NULL);
    assert(ps->size > 0);

    for(int i = 0; i < ps->size - 1; i++)
    {
        ps->a[i] = ps->a[i + 1];
    }

    ps->size--;
}

你会发现：

头插：从后往前移动。

头删：从前往后移动。

尾插：直接在 a[size] 插入。

九、这里还有两个老师代码中值得注意的细节
这部分我希望你不要忽略，因为真正写代码时会遇到。

问题一：capacity 初始为 0 怎么办？

老师写的是：

int newCapacity = ps->capacity * 2;

如果：

capacity = 0;

那么：

newCapacity = 0 * 2;

结果还是 0。

这就无法完成扩容。

所以要么初始化时先申请一定空间，例如 4 个元素；要么扩容时特殊处理：

int newCapacity = ps->capacity == 0
                  ? 4
                  : ps->capacity * 2;

问题二：assert(ps->a != NULL) 是否一定正确？

不一定。

如果初始化时采用：

ps->a = NULL;
ps->size = 0;
ps->capacity = 0;

那么这个断言就会失败。

所以更通用的写法是：

assert(ps != NULL);

然后由扩容函数处理 a == NULL 的情况。

这些属于代码健壮性问题，不影响你理解老师当前采用的设计，但要知道不同设计之间存在区别。
